Two-Step Developer Integration

Step 1: Frontend HTML (Zero Framework Dependency) index.html 
<!-- 1. Include the universal integration script -->
<script src="https://your-domain.com/wa-userauth.js" data-app-key="your_app_key"></script>

<!-- 2. Add the login button anywhere on your site -->
<button data-wa-userauth>Login with WhatsApp</button>

<!-- 3. Listen for the authentication result event -->
<script>
  window.addEventListener('wa-userauth:success', async (e) => {
    const { authToken } = e.detail;
    // Send authToken to your backend for server-to-server validation:
    const res = await fetch('/api/login/verify-wa', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ authToken })
    });
    const session = await res.json();
    console.log('Logged in user:', session.user);
  });
</script>
Step 2: Backend Verification (Node.js / Express Example) server.js 
app.post('/api/login/verify-wa', async (req, res) => {
  const { authToken } = req.body;

  // Validate the single-use token with your WA-USERAUTH instance:
  const verifyRes = await fetch('https://your-domain.com/api/v1/auth/result/verify', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      authToken,
      appKey: process.env.WA_APP_KEY,
      appSecret: process.env.WA_APP_SECRET
    })
  });

  const outcome = await verifyRes.json();
  if (!outcome.valid) {
    return res.status(401).json({ error: outcome.error });
  }

  // Authoritative verified profile: outcome.user { name, mobile, email }
  req.session.user = outcome.user;
  res.json({ success: true, user: outcome.user });
});
